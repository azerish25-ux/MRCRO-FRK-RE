cmake_minimum_required(VERSION 3.22)
project(mfrk_core VERSION 0.1.0 LANGUAGES CXX)

add_library(mfrk_core STATIC core/src/pitch.cpp core/src/virtual_analog.cpp
    core/src/linear_upsampler.cpp core/src/virtual_analog_controls.cpp
    core/src/virtual_analog_oscillator.cpp core/src/basic_wave_tables.cpp
    core/src/basic_waves.cpp core/src/vcf37.cpp core/src/vcf_sample.cpp core/src/vcf49.cpp
    core/src/iir_resampler.cpp core/src/oversampled_vcf49.cpp core/src/adsr.cpp
    core/src/filter_resampling.cpp core/src/fir_resampler.cpp core/src/filter_resource.cpp
    core/src/deferred_vcf49.cpp core/src/superwave.cpp core/src/fx_input.cpp core/src/fx_mix.cpp core/src/bass.cpp core/src/harmo.cpp core/src/notepath.cpp)
target_include_directories(mfrk_core PUBLIC core/include)
target_include_directories(mfrk_core PRIVATE third_party)
# Upstream stmlib's TEST path selects portable helpers instead of Cortex-M assembly.
target_compile_definitions(mfrk_core PRIVATE TEST)
target_compile_features(mfrk_core PUBLIC cxx_std_20)
set_target_properties(mfrk_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
    target_compile_options(mfrk_core PRIVATE /fp:strict)
else()
    target_compile_options(mfrk_core PRIVATE -fno-fast-math -ffp-contract=off)
    target_link_libraries(mfrk_core PRIVATE m)
endif()
