cmake_minimum_required(VERSION 3.22)
project(mfrk_core VERSION 0.1.0 LANGUAGES CXX)

add_library(mfrk_core STATIC core/src/pitch.cpp core/src/virtual_analog.cpp)
target_include_directories(mfrk_core PUBLIC core/include)
target_compile_features(mfrk_core PUBLIC cxx_std_20)
set_target_properties(mfrk_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
    target_compile_options(mfrk_core PRIVATE /fp:strict)
else()
    target_compile_options(mfrk_core PRIVATE -fno-fast-math -ffp-contract=off)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(mfrk_pitch_tests core/tests/pitch_test.cpp)
    target_link_libraries(mfrk_pitch_tests PRIVATE mfrk_core)
    add_test(NAME original_pitch_vectors COMMAND mfrk_pitch_tests)
    add_executable(mfrk_vanalog_tests core/tests/virtual_analog_test.cpp)
    target_link_libraries(mfrk_vanalog_tests PRIVATE mfrk_core)
    add_test(NAME original_vanalog_vectors COMMAND mfrk_vanalog_tests)
endif()
